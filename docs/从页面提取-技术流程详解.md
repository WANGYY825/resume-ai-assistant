# 从页面提取功能 - 技术流程详解

本文档详细说明"从页面提取"功能的完整流程，从点击按钮到数据存储再到复用填写。

---

## 🔄 完整流程概览

```
用户点击按钮
    ↓
扫描页面已填字段
    ↓
发送给 AI 识别
    ↓
AI 返回结构化数据
    ↓
保存到本地存储
    ↓
下次自动填写时复用
```

## 🔍 第二步：提取字段信息

### 2.1 扫描页面表单（content.js）
```javascript
// 文件：extension/content.js
function extractFilledFields() {
  // 1. 查找所有表单元素（input、textarea、select）
  const selectors = "input:not([type=hidden]):not([type=submit]):not([type=button]):not([type=password]), textarea, select";
  
  // 2. 过滤掉隐藏元素
  const elements = Array.from(document.querySelectorAll(selectors)).filter(el => {
    const style = window.getComputedStyle(el);
    return style.display !== 'none' && style.visibility !== 'hidden';
  });

  const filledFields = [];
  
  // 3. 遍历每个元素，提取已填写的值
  elements.forEach(el => {
    let value = "";
    let displayValue = "";

    if (el.tagName === 'SELECT') {
      // 下拉框：提取选中的选项
      const selected = el.options[el.selectedIndex];
      if (selected && selected.value !== '') {
        value = selected.value;
        displayValue = selected.text.trim();  // 显示文字
      }
    } else if (el.type === 'checkbox') {
      // 复选框：提取勾选状态
      if (el.checked) {
        value = 'true';
        displayValue = '已勾选';
      }
    } else if (el.type === 'radio') {
      // 单选框：提取选中的值
      if (el.checked) {
        value = el.value;
        displayValue = el.value;
      }
    } else {
      // 文本框：提取输入内容
      value = el.value?.trim() || '';
      displayValue = value;
    }

    if (value && value !== '') {
      filledFields.push({
        tag: el.tagName.toLowerCase(),
        type: el.type || '',
        name: el.name || '',
        label: getLabel(el),  // 获取字段标签（如"姓名"、"手机号"）
        value: value,
        displayValue: displayValue,
        placeholder: el.placeholder || ''
      });
    }
  });

  return filledFields;
}
```

### 2.2 获取字段标签
```javascript
function getLabel(el) {
  // 方法1：通过 <label for="xxx"> 标签
  if (el.id) {
    const label = document.querySelector(`label[for="${el.id}"]`);
    if (label) return label.textContent.trim();
  }
  
  // 方法2：寻找最近的 <label> 父元素
  let parent = el.parentElement;
  while (parent) {
    const label = parent.querySelector('label');
    if (label) return label.textContent.trim();
    parent = parent.parentElement;
  }
  
  // 方法3：使用 placeholder 或 name 属性
  return el.placeholder || el.name || '';
}
```

### 3.2 调用 Claude API（background.js）
```javascript
// 文件：extension/background.js
if (msg.type === 'FILLED_FIELDS_EXTRACTED') {
  const {filledFields = []} = msg;

  // 1. 格式化字段描述

    `- label: "${f.label}", name: "${f.name}", value: "${f.displayValue}", type: ${f.type}`
  ).join('\n');

  // 示例输出：
  // - label: "姓名", name: "realname", value: "张三", type: text
  // - label: "最高学历", name: "education", value: "硕士研究生", type: select-one
  // - label: "项目经历", name: "project_desc", value: "负责开发AI简历助手...", type: textarea

  // 2. 调用 Claude API

    EXTRACT_SYSTEM,  // AI 系统提示词

    const extracted = tb.input;  // AI 提取的结构化简历数据
    
    // 继续下一步...
  });
}
```

```javascript
chrome.storage.local.get(['profileVersions']).then(store => {
  const versions = store.profileVersions || {};
  
  // 添加新版本
  versions[versionName] = extracted;
  
  // 保存并设为活跃版本
```

```javascript
async function getVersions() {

async function initVersions() {
  const versions = await getVersions();
```

### 6.2 扫描表单字段（content.js）
```javascript
function scanFields() {
  const elements = Array.from(document.querySelectorAll('input, textarea, select'));
  
  return elements.map((el, idx) => {
    el.setAttribute('data-ai-idx', idx);  // 标记索引，方便后续填写
    
    return {
      index: idx,
      tag: el.tagName.toLowerCase(),
      type: el.type || '',
      name: el.name || '',
      placeholder: el.placeholder || '',
      label: getLabel(el),  // 获取字段标签
      selector: `[data-ai-idx="${idx}"]`,
      options: el.tagName === 'SELECT'
        ? Array.from(el.options).map(o => ({value: o.value, text: o.text}))
        : []
    };
  });
}
```

### 6.4 自动填写字段（content.js）
```javascript
if (message.type === 'FILL_FIELDS') {
  let filled = 0;
  
  for (const {selector, value} of message.matches) {
    const el = document.querySelector(selector);
    if (el) {
      fillField(el, value);  // 填写字段
      filled++;
      el.style.background = '#d4edda';  // 高亮显示
    }
  }
  
  console.log(`已填写 ${filled} 个字段`);
}

function fillField(el, value) {
  if (el.tagName === 'SELECT') {
    // 下拉框：匹配文本选中
    const match = Array.from(el.options).find(o =>
      o.text.trim() === value || o.text.includes(value)
    );
    if (match) el.value = match.value;
  } else if (el.type === 'checkbox') {
    // 复选框：勾选
    el.checked = true;
  } else if (el.type === 'radio') {
    // 单选框：选中
    el.checked = true;
  } else {
    // 文本框：填入内容
    el.value = value;
  }
  
  // 触发 input 事件（某些网站需要）
  el.dispatchEvent(new Event('input', {bubbles: true}));
}
```

## 🎯 核心要点

1. **提取**：扫描页面 → 获取已填字段 + 标签
2. **识别**：AI 语义理解 → 映射到标准简历结构
3. **存储**：保存到浏览器本地 → 多版本管理
4. **复用**：下次填表时 AI 匹配 → 自动填写

**优势**：
- ✅ 完全本地存储，隐私安全
- ✅ AI 语义识别，不依赖固定字段名
- ✅ 多版本管理，不同场景灵活切换
- ✅ 双向能力：既能导出（填写），也能导入（提取）

**技术栈**：
- Chrome Extension API（消息通信、本地存储）
- Claude API（语义识别、字段映射）
- 原生 JavaScript（无框架依赖）
